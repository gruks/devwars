  const result = await executeInSandbox({
    language: LANGUAGE_MAP[language] || language,
    code,
    input,
    timeout: Math.min(timeout, 5000)
  });
  
  return {
    success: result.success,
    output: result.output,
    error: result.error,
    runtime: result.executionTime,
    memory: result.memoryUsage
  };
};
